{{/* Person structured data (JSON-LD), driven entirely by data/cv.yaml and site.Params.socialLinks so it can't drift from the CV. Included from head.html on the home and CV pages only. */}} {{- $cv := hugo.Data.cv -}} {{- $lang := site.Language.Lang -}} {{/* Stable, language-independent identity: same @id everywhere so every page describes ONE Person/ProfilePage entity rather than one per language. */}} {{- $canonicalRoot := "" -}} {{- range hugo.Sites -}} {{- if .Language.IsDefault -}}{{- $canonicalRoot = .Home.Permalink -}}{{- end -}} {{- end -}} {{- $personID := printf "%s#person" $canonicalRoot -}} {{- $email := site.Params.contact.email -}} {{- $phone := replace site.Params.contact.phone " " "" -}} {{- $sameAs := slice -}} {{- range site.Params.socialLinks -}} {{- $sameAs = $sameAs | append .url -}} {{- end -}} {{- $knowsLanguage := slice -}} {{- range $cv.spokenLanguages -}} {{- $knowsLanguage = $knowsLanguage | append (index .name $lang) -}} {{- end -}} {{- $alumniOf := slice -}} {{- range $cv.education -}} {{- $alumniOf = $alumniOf | append (dict "@type" "CollegeOrUniversity" "name" .school) -}} {{- end -}} {{- $knowsAbout := slice -}} {{- range $cv.skills -}} {{- $knowsAbout = $knowsAbout | append .items -}} {{- end -}} {{- $knowsAbout = $knowsAbout | uniq -}} {{/* Current employer: the first role with no end date */}} {{- $currentCompany := "" -}} {{- $currentCompanyUrl := "" -}} {{- range $cv.experiences -}} {{- $company := . -}} {{- if eq $currentCompany "" -}} {{- range $company.roles -}} {{- if and (not .end) (eq $currentCompany "") -}} {{- $currentCompany = $company.company -}} {{- $currentCompanyUrl = $company.url -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $person := dict "@type" "Person" "@id" $personID "name" $cv.header.name "url" $canonicalRoot "jobTitle" (index $cv.header.title $lang) "description" (index $cv.header.summary $lang) "sameAs" $sameAs "knowsLanguage" $knowsLanguage "alumniOf" $alumniOf "knowsAbout" $knowsAbout -}} {{- if ne $email "" -}} {{- $person = merge $person (dict "email" $email) -}} {{- end -}} {{- if ne $phone "" -}} {{- $person = merge $person (dict "telephone" $phone) -}} {{- end -}} {{- if ne $currentCompany "" -}} {{- $org := dict "@type" "Organization" "name" $currentCompany -}} {{- if ne $currentCompanyUrl "" -}} {{- $org = merge $org (dict "url" $currentCompanyUrl) -}} {{- end -}} {{- $person = merge $person (dict "worksFor" $org) -}} {{- end -}} {{- $data := dict "@context" "https://schema.org" "@type" "ProfilePage" "@id" .Permalink "url" .Permalink "inLanguage" $lang "dateModified" (.Lastmod.Format "2006-01-02") "mainEntity" $person -}}
